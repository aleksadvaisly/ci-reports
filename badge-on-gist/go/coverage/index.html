import (

type Record map[string]string

func Parse(data string) ([]Record, error) {
        reader := csv.NewReader(strings.NewReader(data))
        reader.FieldsPerRecord = -1

        rows, err := reader.ReadAll()
        if err != nil {
                return nil, fmt.Errorf("parse csv: %w", err)
        }
        if len(rows) == 0 {
                return nil, fmt.Errorf("csv is empty")
        }

        headers := rows[0]
        if len(headers) == 0 {
                return nil, fmt.Errorf("missing headers")
        }

        seen := map[string]struct{}{}
        for _, h := range headers {
                if h == "" {
                        return nil, fmt.Errorf("header cannot be empty")
                }
                if _, ok := seen[h]; ok {
                        return nil, fmt.Errorf("duplicate header: %s", h)
                }
                seen[h] = struct{}{}
        }

        records := make([]Record, 0, len(rows)-1)
        for i, row := range rows[1:] {
                if len(row) != len(headers) {
                        return nil, fmt.Errorf("row %d has %d columns, expected %d", i+2, len(row), len(headers))
                }

                record := make(Record, len(headers))
                for idx, header := range headers {
                        record[header] = row[idx]
                }
                records = append(records, record)
        }

        return records, nil
}

func SumInt(records []Record, column string) (int, error) {
        if err := ensureColumn(records, column); err != nil {
                return 0, err
        }

        total := 0
        for i, record := range records {
                value, err := strconv.Atoi(record[column])
                if err != nil {
                        return 0, fmt.Errorf("row %d column %q is not an int: %w", i+1, column, err)
                }
                total += value
        }

        return total, nil
}

func AverageInt(records []Record, column string) (float64, error) {
        if len(records) == 0 {
                return 0, fmt.Errorf("cannot average empty dataset")
        }

        total, err := SumInt(records, column)
        if err != nil {
                return 0, err
        }

        return float64(total) / float64(len(records)), nil
}

func Filter(records []Record, column, value string) ([]Record, error) {
        if err := ensureColumn(records, column); err != nil {
                return nil, err
        }

        filtered := make([]Record, 0, len(records))
        for _, record := range records {
                if record[column] == value {
                        filtered = append(filtered, record)
                }
        }

        return filtered, nil
}

func ensureColumn(records []Record, column string) error {
        if column == "" {
                return fmt.Errorf("column name cannot be empty")
        }
        if len(records) == 0 {
                return fmt.Errorf("no records available")
        }
        if _, ok := records[0][column]; !ok {
                return fmt.Errorf("column %q does not exist", column)
        }

        return nil
}

func cloneRecord(record Record) Record {
        out := make(Record, len(record))
        for key, value := range record {
                out[key] = value
        }
        return out
}

func cloneRecords(records []Record) []Record {
        out := make([]Record, 0, len(records))
        for _, record := range records {
                out = append(out, cloneRecord(record))
        }
        return out
}

func intsFromColumn(records []Record, column string) ([]int, error) {
        if err := EnsureColumn(records, column); err != nil {
                return nil, err
        }

        values := make([]int, 0, len(records))
        for _, record := range records {
                values = append(values, mustInt(record[column]))
        }
        return values, nil
}

func mustInt(raw string) int {
        value, _ := strconv.Atoi(raw)
        return value
}

func RenderTable(records []Record, columns []string) (string, error) {
        projected, err := Project(records, columns)
        if err != nil {
                return "", err
        }

        lines := make([]string, 0, len(projected)+1)
        lines = append(lines, strings.Join(columns, " | "))
        for _, record := range projected {
                row := make([]string, 0, len(columns))
                for _, column := range columns {
                        row = append(row, record[column])
                }
                lines = append(lines, strings.Join(row, " | "))
        }
        return strings.Join(lines, "\n"), nil
}

}

func SortByInt(records []Record, column string, desc bool) ([]Record, error) {
        if err := EnsureColumn(records, column); err != nil {
                return nil, err
        }

        sorted := cloneRecords(records)
        sort.Slice(sorted, func(i, j int) bool {
                left := mustInt(sorted[i][column])
                right := mustInt(sorted[j][column])
                if desc {
                        return left > right
                }
                return left < right
        })
        return sorted, nil
}

import "strings"

func splitLines(data string) []string {
        lines := strings.Split(strings.ReplaceAll(data, "\r\n", "\n"), "\n")
        out := make([]string, 0, len(lines))
        for _, line := range lines {
                if strings.TrimSpace(line) == "" {
                        continue
                }
                out = append(out, line)
        }
        return out
}

func splitFields(line string) []string {
        parts := strings.Split(line, ",")
        for i := range parts {
                parts[i] = strings.TrimSpace(parts[i])
        }
        return parts
}
